  const outlinedBtnSx = {
    height: 34,
    textTransform: "none",
    fontWeight: 650,
    fontSize: "12.5px",
    borderRadius: "8px",
    px: 1.75,
    borderColor: "#c7d2e0",
    color: "#334155",
    background: "linear-gradient(180deg, #ffffff 0%, #f8fafc 100%)",
    boxShadow: "0 1px 2px rgba(15, 23, 42, 0.06)",
    "&:hover": {
      borderColor: "#16408f",
      color: "#16408f",
      background: "#eff6ff",
      boxShadow: "0 4px 12px rgba(22, 64, 143, 0.12)",
    },
  };

  return (
    <div className="ff-module">
      <div className="ff-header">
        <div className="ff-header-title">
          <h1>Freight Forwarding</h1>
          <p>View enquiries, quotations, shipments, and historic freight rates</p>
        </div>
        <div className="ff-header-actions">
          <Button
            variant="outlined"
            size="small"
            startIcon={<DownloadIcon sx={{ fontSize: 16 }} />}
            onClick={() => {
              setDsrMode("Import");
              setDsrShipmentType("all");
              setOpenDSRDialog(true);
            }}
            sx={outlinedBtnSx}
          >
            Import DSR
          </Button>
          <Button
            variant="outlined"
            size="small"
            startIcon={<DownloadIcon sx={{ fontSize: 16 }} />}
            onClick={() => {
              setDsrMode("Export");
              setDsrShipmentType("all");
              setOpenDSRDialog(true);
            }}
            sx={outlinedBtnSx}
          >
            Export DSR
          </Button>
        </div>
      </div>

      <div className="ff-tabs-shell">
        {visibleTabs.map((tab) => {
          const isActive = activeTab === tab.key;
          return (
            <button
              key={tab.key}
              type="button"
              className={`ff-tab${isActive ? " is-active" : ""}`}
              onClick={() => setActiveTab(tab.key)}
            >
              {tab.key}
              {tab.count !== undefined && <span className="ff-tab-badge">{tab.count}</span>}
            </button>
          );
        })}
      </div>

      {activeTab === "Historic Rates" ? (
        <HistoricRatesLookup />
      ) : (
        <>
          <div className="ff-toolbar">
            <input
              value={filters.search}
              onChange={(e) => setFilters((prev) => ({ ...prev, search: e.target.value }))}
              onFocus={() => setSearchFocused(true)}
              onBlur={() => setSearchFocused(false)}
              placeholder="Search by Enquiry No, Org, Port..."
              className={`ff-search${searchFocused || filters.search ? " is-active" : ""}`}
            />
            <select
              value={filters.shipment_type}
              onChange={(e) => setFilters((prev) => ({ ...prev, shipment_type: e.target.value }))}
              className={`ff-select${filters.shipment_type ? " is-active" : ""}`}
            >
              <option value="">All Shipment Types</option>
              <option value="Import-Sea">Import - Sea</option>
              <option value="Export-Sea">Export - Sea</option>
              <option value="Import-Air">Import - Air</option>
              <option value="Export-Air">Export - Air</option>
            </select>
          </div>

          <div className="ff-table-card">
            <div className="ff-table-wrap">
              <table className="ff-table">
                <thead>
                  <tr>
                    {[
                      activeTab === "Success" ? "Success No" : activeTab === "Rejected" ? "Rejected No" : "Enquiry No",
                      "Organization Details",
                      "Port & Routing",
                      "Container & Cargo Details",
                      "Tracking & Timeline",
                      "Actions",
                    ].map((h) => (
                      <th key={h} className={h === "Actions" ? "ff-th-actions" : undefined}>
                        {h}
                      </th>
                    ))}
                  </tr>
                </thead>
                <tbody>
                  {loading ? (
                    <tr>
                      <td colSpan={6} className="ff-empty">
                        Loading...
                      </td>
                    </tr>
                  ) : filteredRows.length ? (
                    filteredRows.map((row) => (
                      <tr
                        key={row._id || row.enquiry_no}
                        className={activeTab !== "Success" ? "is-clickable" : undefined}
                        onClick={() => handleRowClick(row)}
                      >
                        <td>
                          <div style={{ display: "flex", flexDirection: "column", gap: "5px" }}>
                            <span className="ff-id">
                              {activeTab === "Success"
                                ? row.success_no || row.enquiry_no
                                : activeTab === "Rejected"
                                  ? row.rejected_no || row.enquiry_no
                                  : row.enquiry_no}
                            </span>
                            <div className="ff-meta">Date: {row.enquiry_date}</div>
                            {row.source_job_no && <span className="ff-chip-ref">Ref Job: {row.source_job_no}</span>}
                            {(row.shipment_ref_no || row.bl_details?.shipment_ref_no) && (
                              <span className="ff-chip ff-chip-teal">
                                Ref: {row.shipment_ref_no || row.bl_details?.shipment_ref_no}
                              </span>
                            )}
                            <span className="ff-chip ff-chip-blue">{row.shipment_type}</span>
                          </div>
                        </td>
                        <td>
                          <div style={{ fontWeight: 700, fontSize: "12.5px", color: "#0f172a" }}>
                            {String(row.shipment_type || "").startsWith("Import") ? "Consignee: " : "Shipper: "}
                            {String(row.shipment_type || "").startsWith("Import")
                              ? row.consignee_name || row.bl_details?.consignee || row.organization_name || "-"
                              : row.shipper_name || row.organization_name || "-"}
                          </div>
                          {row.email && (
                            <div className="ff-meta">
                              <strong>Email:</strong> {row.email}
                            </div>
                          )}
                          {row.contact_no && (
                            <div className="ff-meta">
                              <strong>Contact:</strong> {row.contact_no}
                            </div>
                          )}
                        </td>
                        <td>
                          <div style={{ display: "flex", flexDirection: "column", gap: "5px" }}>
                            <div style={{ display: "flex", gap: 4 }}>
                              <span className="ff-route-label">Place of Receipt:</span>
                              <span className="ff-route-value">
                                {row.place_of_receipt || row.bl_details?.place_of_acceptance || "-"}
                              </span>
                            </div>
                            <div style={{ display: "flex", gap: 4 }}>
                              <span className="ff-route-label">POL:</span>
                              <span className="ff-route-value">{row.port_of_loading || "-"}</span>
                            </div>
                            <div style={{ display: "flex", gap: 4 }}>
                              <span className="ff-route-label">POD:</span>
                              <span className="ff-route-value">{row.port_of_destination || "-"}</span>
                            </div>
                            {row.bl_details?.vessel_name && (
                              <div className="ff-meta" style={{ marginTop: 4 }}>
                                <strong>Vessel:</strong> {row.bl_details.vessel_name}{" "}
                                {row.bl_details.voyage_no ? `V ${row.bl_details.voyage_no}` : ""}
                              </div>
                            )}
                          </div>
                        </td>
                        <td>
                          <div style={{ display: "flex", gap: 4, flexWrap: "wrap", marginBottom: 6 }}>
                            {row.container_size && <span className="ff-chip ff-chip-slate">{row.container_size}</span>}
                            {row.consignment_type && (
                              <span className="ff-chip ff-chip-slate">{row.consignment_type}</span>
                            )}
                            {row.goods_stuffed && <span className="ff-chip ff-chip-slate">{row.goods_stuffed}</span>}
                          </div>
                          <div className="ff-cargo-box">
                            <div className="ff-cargo-row">
                              <span>Pkgs:</span>
                              <span>
                                {row.no_packages || "-"} {row.package_unit || ""}
                              </span>
                            </div>
                            <div className="ff-cargo-row">
                              <span>Gross Wt:</span>
                              <span>
                                {row.gross_weight || "-"} {row.gross_weight_unit || ""}
                              </span>
                            </div>
                            <div className="ff-cargo-row">
                              <span>Net Wt:</span>
                              <span>
                                {row.net_weight || "-"} {row.net_weight_unit || ""}
                              </span>
                            </div>
                            {row.volume_cbm && (
                              <div className="ff-cargo-row">
                                <span>Volume:</span>
                                <span>
                                  {row.volume_cbm} {row.volume_unit || "CBM"}
                                </span>
                              </div>
                            )}
                          </div>
                          {row.containers?.some((c) => c.container_number) && (
                            <div style={{ marginTop: 6 }}>
                              {row.containers
                                .filter((c) => c.container_number)
                                .map((c, i) => (
                                  <div key={i} className="ff-id" style={{ fontSize: "10.5px" }}>
                                    {c.container_number}
                                  </div>
                                ))}
                            </div>
                          )}
                        </td>
                        <td>
                          <div style={{ display: "flex", justifyContent: "space-between", fontSize: "11px" }}>
                            <span className="ff-timeline-label">E.T.A:</span>
                            <span className="ff-timeline-value">
                              {row.eta_date ? row.eta_date.split("-").reverse().join("/") : "-"}
                            </span>
                          </div>
                          <div
                            style={{
                              display: "flex",
                              justifyContent: "space-between",
                              fontSize: "11px",
                              marginTop: 4,
                            }}
                          >
                            <span className="ff-timeline-label">ARRIVAL:</span>
                            <span className="ff-timeline-value">
                              {row.arrival_date ? row.arrival_date.split("-").reverse().join("/") : "-"}
                            </span>
                          </div>
                          {row.delay_reason && (
                            <div
                              style={{
                                marginTop: 6,
                                background: "linear-gradient(180deg, #fffbeb, #fef3c7)",
                                border: "1px solid #fde68a",
                                padding: "5px 7px",
                                borderRadius: 8,
                              }}
                            >
                              <span style={{ fontSize: 8.5, fontWeight: 800, color: "#d97706" }}>Delay Reason:</span>
                              <div style={{ fontSize: 10, fontWeight: 600, color: "#b45309" }}>{row.delay_reason}</div>
                            </div>
                          )}
                        </td>
                        <td>
                          <div className="ff-actions">
                            <DocsViewCell row={row} />
                            {row.status === "Converted" && (
                              <Tooltip title="Track Shipment">
                                <IconButton
                                  size="small"
                                  onClick={(e) => {
                                    e.stopPropagation();
                                    setTrackingEnquiry(row);
                                  }}
                                  sx={{
                                    border: "1px solid #fed7aa",
                                    background: "linear-gradient(180deg, #fff7ed, #ffedd5)",
                                    color: "#ea580c",
                                    borderRadius: "8px",
                                    boxShadow: "0 1px 3px rgba(234, 88, 12, 0.15)",
                                    "&:hover": {
                                      background: "#ffedd5",
                                      borderColor: "#fb923c",
                                      transform: "translateY(-1px)",
                                    },
                                  }}
                                >
                                  <LocalShippingIcon fontSize="small" />
                                </IconButton>
                              </Tooltip>
                            )}
                          </div>
                        </td>
                      </tr>
                    ))
                  ) : (
                    <tr>
                      <td colSpan={6} className="ff-empty">
                        No {activeTab.toLowerCase()} records found.
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </div>
        </>
      )}

      <Dialog
        open={!!selectedEnquiry}
        onClose={() => setSelectedEnquiry(null)}
        maxWidth="md"
        fullWidth
        sx={{ "& .MuiDialog-paper": { borderRadius: "12px", overflow: "hidden" } }}
      >
        <DialogTitle
          sx={{
            m: 0,
            p: 2,
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            background: "linear-gradient(135deg, #12367a, #16408f)",
            color: "#fff",
          }}
        >
          <Typography sx={{ fontWeight: 700, fontSize: "14px" }}>
            Enquiry Details - {selectedEnquiry?.enquiry_no}
          </Typography>
          <IconButton onClick={() => setSelectedEnquiry(null)} sx={{ color: "#fff" }}>
            <CloseIcon fontSize="small" />
          </IconButton>
        </DialogTitle>
        <DialogContent sx={{ p: 0 }}>{selectedEnquiry && <CaptureRates enquiry={selectedEnquiry} />}</DialogContent>
      </Dialog>

      {trackingEnquiry && (
        <FreightTrackingMap enquiry={trackingEnquiry} onClose={() => setTrackingEnquiry(null)} />
      )}

      <Dialog
        open={openDSRDialog}
        onClose={() => {
          setOpenDSRDialog(false);
          setDsrStartDate("");
          setDsrEndDate("");
        }}
        maxWidth="xs"
        fullWidth
        PaperProps={{ style: { borderRadius: "12px" } }}
      >
        <DialogTitle
          sx={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            pb: 1,
            borderBottom: "1px solid #e2e8f0",
          }}
        >
          <span style={{ fontSize: "15px", fontWeight: "700", color: "#1e293b" }}>
            Freight Forwarding {dsrMode} DSR Report
          </span>
          <IconButton
            size="small"
            onClick={() => {
              setOpenDSRDialog(false);
              setDsrStartDate("");
              setDsrEndDate("");
            }}
            sx={{ color: "#64748b" }}
          >
            <CloseIcon fontSize="small" />
          </IconButton>
        </DialogTitle>
        <DialogContent sx={{ pt: 2, pb: 2 }}>
          <Box sx={{ display: "flex", flexDirection: "column", gap: 2, mt: 1 }}>
            <FormControl size="small" fullWidth>
              <InputLabel>Financial Year</InputLabel>
              <Select
                value={dsrYear}
                label="Financial Year"
                onChange={(e) => setDsrYear(e.target.value)}
                sx={{ borderRadius: "8px", fontSize: "12px" }}
              >
                <MenuItem value="all" style={{ fontSize: "12px" }}>All Years</MenuItem>
                <MenuItem value="26-27" style={{ fontSize: "12px" }}>2026-2027 (26-27)</MenuItem>
                <MenuItem value="25-26" style={{ fontSize: "12px" }}>2025-2026 (25-26)</MenuItem>
                <MenuItem value="24-25" style={{ fontSize: "12px" }}>2024-2025 (24-25)</MenuItem>
              </Select>
            </FormControl>
            <FormControl size="small" fullWidth>
              <InputLabel>Shipment Type</InputLabel>
              <Select
                value={dsrShipmentType}
                label="Shipment Type"
                onChange={(e) => setDsrShipmentType(e.target.value)}
                sx={{ borderRadius: "8px", fontSize: "12px" }}
              >
                {dsrMode === "Import" ? (
                  <>
                    <MenuItem value="all" style={{ fontSize: "12px" }}>All Import Shipments</MenuItem>
                    <MenuItem value="Import-Sea" style={{ fontSize: "12px" }}>Import - Sea</MenuItem>
                    <MenuItem value="Import-Air" style={{ fontSize: "12px" }}>Import - Air</MenuItem>
                  </>
                ) : (
                  <>
                    <MenuItem value="all" style={{ fontSize: "12px" }}>All Export Shipments</MenuItem>
                    <MenuItem value="Export-Sea" style={{ fontSize: "12px" }}>Export - Sea</MenuItem>
                    <MenuItem value="Export-Air" style={{ fontSize: "12px" }}>Export - Air</MenuItem>
                  </>
                )}
              </Select>
            </FormControl>
            <Box sx={{ display: "flex", gap: 1 }}>
              <TextField
                label="Start Date"
                type="date"
                size="small"
                value={dsrStartDate}
                onChange={(e) => setDsrStartDate(e.target.value)}
                InputLabelProps={{ shrink: true }}
                fullWidth
                inputProps={{ style: { fontSize: "12px" } }}
              />
              <TextField
                label="End Date"
                type="date"
                size="small"
                value={dsrEndDate}
                onChange={(e) => setDsrEndDate(e.target.value)}
                InputLabelProps={{ shrink: true }}
                fullWidth
                inputProps={{ style: { fontSize: "12px" } }}
              />
            </Box>
          </Box>
        </DialogContent>
        <DialogActions sx={{ px: 3, pb: 2 }}>
          <Button
            onClick={handleDownloadDSR}
            disabled={dsrLoading}
            variant="contained"
            sx={{
              background: "linear-gradient(135deg, #16408f, #1e4fa8)",
              textTransform: "none",
              fontWeight: 700,
              fontSize: "12px",
              borderRadius: "8px",
              boxShadow: "0 4px 12px rgba(22, 64, 143, 0.25)",
              "&:hover": { background: "#19448a" },
            }}
          >
            {dsrLoading ? "Downloading..." : "Download Report"}
          </Button>
        </DialogActions>
      </Dialog>
    </div>
  );
}

export default CFreightForwarding;
